\section{admin\_reviews and audit\_logs}
\begin{itemize}
\item \\texttt{admin\\_reviews} stores reviewer decision actions with reason field constraints.
\item \\texttt{audit\\_logs} stores actor/action metadata for traceability and operational review.
\end{itemize}

\section{Important SQL Constraints}
\begin{enumerate}
\item unique user email and phone
\item score value range checks in verification sessions
\item rejection reason required when decision is rejected
\item referential integrity across users, documents, sessions, and reviews
\end{enumerate}
